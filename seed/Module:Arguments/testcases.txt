local Arguments = require('Module:Arguments')
local ScribuntoUnit = require('Module:ScribuntoUnit')

local suite = ScribuntoUnit.new()

function suite:testTrimsAndRemovesBlankArguments()
	local args = Arguments.getArgs({ args = { label = '  Example  ', empty = '   ' } })
	self:assertEquals('Example', args.label)
	self:assertEquals(nil, args.empty)
end

function suite:testPrefersParentArgumentsWhenRequested()
	local args = Arguments.getArgs({
		args = { label = 'module' },
		getParent = function()
			return { args = { label = 'template' } }
		end,
	}, { parentFirst = true })
	self:assertEquals('template', args.label)
end

return suite
